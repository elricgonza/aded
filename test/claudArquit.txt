sql-- Vista: Estudiantes inscritos con información completa
CREATE OR REPLACE VIEW v_estudiantes_detalle AS
SELECT 
    e.id,
    e.nombre || ' ' || e.paterno || ' ' || COALESCE(e.materno, '') as nombre_completo,
    e.ci,
    g.nivel,
    g.grado,
    c.paralelo,
    c.gestion,
    i.inscrito,
    i.descu as descuento,
    i.abandono
FROM est e
INNER JOIN ins i ON e.id = i.est_id
INNER JOIN cur c ON i.cur_id = c.id
INNER JOIN gra g ON c.gra_id = g.id
WHERE e.activo = TRUE;

-- Vista: Calificaciones con detalles
CREATE OR REPLACE VIEW v_calificaciones_completas AS
SELECT 
    e.nombre || ' ' || e.paterno as estudiante,
    m.materia,
    g.nivel || ' ' || g.grado || ' ' || c.paralelo as curso,
    cal.nota1,
    cal.nota2,
    cal.nota3,
    cal.nota_final,
    cal.aprobado
FROM cal
INNER JOIN ins i ON cal.ins_id = i.id
INNER JOIN est e ON i.est_id = e.id
INNER JOIN mat m ON cal.mat_id = m.id
INNER JOIN cur c ON i.cur_id = c.id
INNER JOIN gra g ON c.gra_id = g.id;

-- Vista: Estado de pagos
CREATE OR REPLACE VIEW v_estado_pagos AS
SELECT 
    e.nombre || ' ' || e.paterno as estudiante,
    c.curso || ' ' || c.paralelo as curso,
    COUNT(p.id) as total_cuotas,
    COUNT(CASE WHEN p.pagado THEN 1 END) as cuotas_pagadas,
    COUNT(CASE WHEN NOT COALESCE(p.pagado, FALSE) THEN 1 END) as cuotas_pendientes,
    SUM(p.cuota) as monto_total,
    SUM(CASE WHEN p.pagado THEN p.cuota ELSE 0 END) as monto_pagado,
    SUM(CASE WHEN NOT COALESCE(p.pagado, FALSE) THEN p.cuota ELSE 0 END) as saldo_pendiente
FROM ins i
INNER JOIN est e ON i.est_id = e.id
INNER JOIN cur c ON i.cur_id = c.id
LEFT JOIN pag p ON i.id = p.ins_id
GROUP BY e.id, e.nombre, e.paterno, c.curso, c.paralelo;

-- Vista: Carga académica profesores
CREATE OR REPLACE VIEW v_carga_profesores AS
SELECT 
    p.nombre || ' ' || p.paterno as profesor,
    p.formacion,
    COUNT(DISTINCT a.cur_id) as total_cursos,
    COUNT(DISTINCT a.mat_id) as total_materias,
    STRING_AGG(DISTINCT m.materia, ', ') as materias
FROM pro p
INNER JOIN asi a ON p.id = a.pro_id
INNER JOIN mat m ON a.mat_id = m.id
WHERE p.activo = TRUE
GROUP BY p.id, p.nombre, p.paterno, p.formacion;

sql-- Función: Generar plan de pagos automáticamente
CREATE OR REPLACE FUNCTION fn_generar_plan_pagos(p_ins_id INTEGER)
RETURNS INTEGER AS $$
DECLARE
    v_cur_id INTEGER;
    v_nro_cuotas SMALLINT;
    v_monto_cuota NUMERIC(10,2);
    v_descuento SMALLINT;
    v_monto_final NUMERIC(10,2);
    v_registros INTEGER := 0;
    v_i INTEGER;
BEGIN
    -- Obtener datos de la inscripción
    SELECT i.cur_id, i.descu, co.nro_cuota, co.cuota
    INTO v_cur_id, v_descuento, v_nro_cuotas, v_monto_cuota
    FROM ins i
    INNER JOIN cos co ON i.cur_id = co.cur_id
    WHERE i.id = p_ins_id;
    
    -- Calcular monto con descuento
    v_monto_final := v_monto_cuota - (v_monto_cuota * v_descuento / 100.0);
    
    -- Generar cuotas
    FOR v_i IN 1..v_nro_cuotas LOOP
        INSERT INTO pag (ins_id, nro_cuota, cuota, pagado, creado, act)
        VALUES (p_ins_id, v_i, v_monto_final, FALSE, CURRENT_DATE, CURRENT_DATE);
        
        v_registros := v_registros + 1;
    END LOOP;
    
    RETURN v_registros;
END;
$$ LANGUAGE plpgsql;

-- Función: Calcular nota final
CREATE OR REPLACE FUNCTION fn_calcular_nota_final()
RETURNS TRIGGER AS $$
BEGIN
    NEW.nota_final := (NEW.nota1 + NEW.nota2 + NEW.nota3) / 3.0;
    NEW.aprobado := (NEW.nota_final >= NEW.nota_aprob);
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_cal_calcular_nota
    BEFORE INSERT OR UPDATE ON cal
    FOR EACH ROW
    EXECUTE FUNCTION fn_calcular_nota_final();

python# app/models/estudiante.py
from app.extensions import db
from datetime import datetime

class Estudiante(db.Model):
    __tablename__ = 'est'
    
    id = db.Column(db.Integer, primary_key=True)
    nombre = db.Column(db.String(100), nullable=False)
    paterno = db.Column(db.String(100))
    materno = db.Column(db.String(100))
    nacimiento = db.Column(db.Date)
    masculino = db.Column(db.Boolean, nullable=False)
    ci = db.Column(db.Integer)
    direccion = db.Column(db.String(150))
    activo = db.Column(db.Boolean, nullable=False, default=True)
    obs = db.Column(db.String(100))
    creado = db.Column(db.Date, nullable=False, default=datetime.utcnow)
    act = db.Column(db.Date, nullable=False, default=datetime.utcnow, onupdate=datetime.utcnow)
    
    # Relaciones
    inscripcion = db.relationship('Inscripcion', backref='estudiante', uselist=False)
    
    @property
    def nombre_completo(self):
        return f"{self.nombre} {self.paterno} {self.materno or ''}".strip()
    
    @property
    def edad(self):
        if self.nacimiento:
            today = datetime.today()
            return today.year - self.nacimiento.year - (
                (today.month, today.day) < (self.nacimiento.month, self.nacimiento.day)
            )
        return None
    
    def __repr__(self):
        return f'<Estudiante {self.nombre_completo}>'

python# app/blueprints/estudiantes/routes.py
from flask import Blueprint, render_template, request, redirect, url_for, flash
from flask_login import login_required
from app.models.estudiante import Estudiante
from app.models.inscripcion import Inscripcion
from app.forms.estudiante_forms import EstudianteForm
from app.services.estudiante_service import EstudianteService
from app.blueprints.auth.decorators import permission_required
from app.extensions import db

estudiantes_bp = Blueprint('estudiantes', __name__, url_prefix='/estudiantes')

@estudiantes_bp.route('/')
@login_required
@permission_required('ver_estudiantes')
def lista():
    """Lista de estudiantes con búsqueda y paginación"""
    page = request.args.get('page', 1, type=int)
    search = request.args.get('search', '', type=str)
    
    query = Estudiante.query.filter_by(activo=True)
    
    if search:
        query = query.filter(
            db.or_(
                Estudiante.nombre.ilike(f'%{search}%'),
                Estudiante.paterno.ilike(f'%{search}%'),
                Estudiante.ci.ilike(f'%{search}%')
            )
        )
    
    estudiantes = query.paginate(page=page, per_page=20, error_out=False)
    
    # Si es petición HTMX, solo devolver la tabla
    if request.headers.get('HX-Request'):
        return render_template('estudiantes/tabla_estudiantes.html', 
                             estudiantes=estudiantes, 
                             search=search)
    
    return render_template('estudiantes/lista.html', 
                         estudiantes=estudiantes, 
                         search=search)

@estudiantes_bp.route('/crear', methods=['GET', 'POST'])
@login_required
@permission_required('crear_estudiantes')
def crear():
    """Crear nuevo estudiante"""
    form = EstudianteForm()
    
    if form.validate_on_submit():
        estudiante = EstudianteService.crear_estudiante(form)
        flash(f'Estudiante {estudiante.nombre_completo} creado exitosamente', 'success')
        return redirect(url_for('estudiantes.lista'))
    
    return render_template('estudiantes/crear.html', form=form)

@estudiantes_bp.route('/<int:id>')
@login_required
@permission_required('ver_estudiantes')
def detalle(id):
    """Ver detalle de estudiante"""
    estudiante = Estudiante.query.get_or_404(id)
    return render_template('estudiantes/detalle.html', estudiante=estudiante)

@estudiantes_bp.route('/<int:id>/editar', methods=['GET', 'POST'])
@login_required
@permission_required('editar_estudiantes')
def editar(id):
    """Editar estudiante"""

    form = EstudianteForm(obj=estudiante)

from app.models.estudiante import Estudiante
from app.extensions import db
from datetime import datetime

class EstudianteService:
    
    @staticmethod
    def crear_estudiante(form):
        """Crear un nuevo estudiante"""
        estudiante = Estudiante(
            nombre=form.nombre.data,
            paterno=form.paterno.data,
            materno=form.materno.data,
            nacimiento=form.nacimiento.data,
            masculino=form.masculino.data,
            ci=form.ci.data,
            direccion=form.direccion.data,
            activo=True,
            obs=form.obs.data
        )
        
        db.session.add(estudiante)
        db.session.commit()
        
        return estudiante
    
    @staticmethod
    def actualizar_estudiante(estudiante, form):
        """Actualizar estudiante existente"""
        estudiante.nombre = form.nombre.data
        estudiante.paterno = form.paterno.data
        estudiante.materno = form.materno.data
        estudiante.nacimiento = form.nacimiento.data
        estudiante.masculino = form.masculino.data
        estudiante.ci = form.ci.data
        estudiante.direccion = form.direccion.data
        estudiante.obs = form.obs.data
        estudiante.act = datetime.utcnow()
        
        db.session.commit()
        
        return estudiante
    
    @staticmethod
    def buscar_estudiantes(criterio):
        """Buscar estudiantes por nombre o CI"""
        return Estudiante.query.filter(
            db.or_(
                Estudiante.nombre.ilike(f'%{criterio}%'),
                Estudiante.paterno.ilike(f'%{criterio}%'),
                Estudiante.ci.ilike(f'%{criterio}%')
            )
        ).filter_by(activo=True).all()

from functools import wraps
from flask import abort, flash, redirect, url_for
from flask_login import current_user

def permission_required(*permisos):
    """Decorador para verificar permisos"""
    def decorator(f):
        @wraps(f)
        def decorated_function(*args, **kwargs):
            if not current_user.is_authenticated:
                flash('Debes iniciar sesión', 'warning')
                return redirect(url_for('auth.login'))
            
            if not any(current_user.tiene_permiso(p) for p in permisos):
                flash('No tienes permisos para esta acción', 'danger')
                abort(403)
            
            return f(*args, **kwargs)
        return decorated_function
    return decorator

from app.extensions import db
from flask_login import UserMixin
from werkzeug.security import generate_password_hash, check_password_hash

class Usuario(UserMixin, db.Model):
    __tablename__ = 'usr'
    
    id = db.Column(db.Integer, primary_key=True)
    usuario = db.Column(db.String(80), unique=True, nullable=False)
    email = db.Column(db.String(120), unique=True, nullable=False)
    password = db.Column(db.String(255), nullable=False)
    activo = db.Column(db.Boolean, default=True)
    
    # Relaciones
    roles = db.relationship('Rol', secondary='usr_rol', backref='usuarios')
    
    def set_password(self, password):
        self.password = generate_password_hash(password)
    
    def check_password(self, password):
        return check_password_hash(self.password, password)
    
    def tiene_permiso(self, permiso):
        """Verifica si el usuario tiene un permiso específico"""
        for rol in self.roles:
            if any(p.permiso == permiso for p in rol.permisos):
                return True
