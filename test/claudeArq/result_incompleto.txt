CREATE OR REPLACE VIEW v_estudiantes_activos AS
SELECT 
    e.id,
    e.nombre || ' ' || COALESCE(e.paterno, '') || ' ' || COALESCE(e.materno, '') as nombre_completo,
    e.ci,
    c.paralelo,
    g.grado,
    g.nivel,
    i.descu as descuento
FROM est e
INNER JOIN ins i ON e.id = i.est_id
INNER JOIN cur c ON i.cur_id = c.id
INNER JOIN gra g ON c.gra_id = g.id
WHERE e.activo = true AND i.inscrito = true AND i.abandono = false;

CREATE OR REPLACE VIEW v_resumen_pagos AS
SELECT 
    i.id as inscripcion_id,
    e.nombre || ' ' || e.paterno as estudiante,
    COUNT(p.id) as total_cuotas,
    COUNT(CASE WHEN p.pagado = true THEN 1 END) as cuotas_pagadas,
    COUNT(CASE WHEN p.pagado = false OR p.pagado IS NULL THEN 1 END) as cuotas_pendientes,
    SUM(p.cuota) as monto_total,
    SUM(CASE WHEN p.pagado = true THEN p.cuota ELSE 0 END) as monto_pagado,
    SUM(CASE WHEN p.pagado = false OR p.pagado IS NULL THEN p.cuota ELSE 0 END) as monto_pendiente
FROM ins i
INNER JOIN est e ON i.est_id = e.id
LEFT JOIN pag p ON i.id = p.ins_id
GROUP BY i.id, e.nombre, e.paterno;

-- FUNCIÓN: Generar plan de pagos automático
CREATE OR REPLACE FUNCTION fn_generar_plan_pagos(p_ins_id INTEGER)
RETURNS INTEGER AS $$
DECLARE
    v_cur_id INTEGER;
    v_cos_record RECORD;
    v_descu SMALLINT;
    v_cuota_final NUMERIC(10,2);
    v_registros INTEGER := 0;
BEGIN
    -- Obtener datos de inscripción
    SELECT cur_id, descu INTO v_cur_id, v_descu
    FROM ins WHERE id = p_ins_id;
    
    -- Obtener configuración de costos
    SELECT * INTO v_cos_record FROM cos WHERE cur_id = v_cur_id;
    
    -- Aplicar descuento
    v_cuota_final := v_cos_record.cuota * (1 - v_descu / 100.0);
    
    -- Generar cuotas
    FOR i IN 1..v_cos_record.nro_cuota LOOP
        INSERT INTO pag (ins_id, nro_cuota, cuota, pagado, creado, act)
        VALUES (p_ins_id, i, v_cuota_final, false, CURRENT_DATE, CURRENT_DATE);
        v_registros := v_registros + 1;
    END LOOP;
    
    RETURN v_registros;
END;
$$ LANGUAGE plpgsql;

-- TRIGGER: Actualizar campo 'act' automáticamente
CREATE OR REPLACE FUNCTION actualizar_fecha_act()
RETURNS TRIGGER AS $$
BEGIN
    NEW.act = CURRENT_DATE;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_est_act BEFORE UPDATE ON est
    FOR EACH ROW EXECUTE FUNCTION actualizar_fecha_act();
    
CREATE TRIGGER trg_ins_act BEFORE UPDATE ON ins
    FOR EACH ROW EXECUTE FUNCTION actualizar_fecha_act();

"""
Inicialización de la aplicación Flask
"""
from flask import Flask
from flask_sqlalchemy import SQLAlchemy
from flask_login import LoginManager
from flask_bcrypt import Bcrypt
from config import Config

db = SQLAlchemy()
login_manager = LoginManager()
bcrypt = Bcrypt()

def create_app(config_class=Config):
    app = Flask(__name__)
    app.config.from_object(config_class)
    
    # Inicializar extensiones
    db.init_app(app)
    login_manager.init_app(app)
    bcrypt.init_app(app)
    
    login_manager.login_view = 'auth.login'
    login_manager.login_message = 'Por favor inicie sesión'
    
    # Registrar blueprints
    from app.auth import bp as auth_bp
    from app.academico import bp as academico_bp
    from app.financiero import bp as financiero_bp
    from app.reportes import bp as reportes_bp
    
    app.register_blueprint(auth_bp, url_prefix='/auth')
    app.register_blueprint(academico_bp, url_prefix='/academico')
    app.register_blueprint(financiero_bp, url_prefix='/financiero')
    app.register_blueprint(reportes_bp, url_prefix='/reportes')
    
    return app

        for rol in self.roles:
            if any(p.permiso == permiso_nombre for p in rol.permisos):

        return any(r.rol == rol_nombre for r in self.roles)

"""
Servicios de lógica de negocio para el módulo académico
"""
from app import db
from app.models import Estudiante, Inscripcion, Curso, Calificacion
from sqlalchemy import func

class AcademicoService:
    
    @staticmethod
    def inscribir_estudiante(est_id, cur_id, descuento=0, motivo_descu=None):
        """Inscribe un estudiante en un curso"""
        # Verificar capacidad del curso
        curso = Curso.query.get_or_404(cur_id)
        inscritos = Inscripcion.query.filter_by(cur_id=cur_id, inscrito=True).count()
        
        if curso.capacidad and inscritos >= curso.capacidad:
            raise ValueError(f"El curso {curso.paralelo} está lleno")
        
        # Verificar inscripción previa
        inscripcion_existente = Inscripcion.query.filter_by(est_id=est_id).first()
        if inscripcion_existente and inscripcion_existente.inscrito:
            raise ValueError("El estudiante ya está inscrito en un curso")
        
        # Crear inscripción
        inscripcion = Inscripcion(
            est_id=est_id,
            cur_id=cur_id,
            inscrito=True,
            descu=descuento,
            motivo_descu=motivo_descu
        )
        
        db.session.add(inscripcion)
        db.session.commit()
        
        return inscripcion
    
    @staticmethod
    def registrar_calificaciones(ins_id, mat_id, nota1, nota2, nota3, nota_aprob=51):
        """Registra calificaciones de un estudiante en una materia"""
        nota_final = (nota1 + nota2 + nota3) / 3.0
        aprobado = nota_final >= nota_aprob
        
        calificacion = Calificacion(
            ins_id=ins_id,
            mat_id=mat_id,
            nota1=nota1,
            nota2=nota2,
            nota3=nota3,
            nota_final=nota_final,
            nota_aprob=nota_aprob,
            aprobado=aprobado
        )
        
        db.session.add(calificacion)
        db.session.commit()
        
        return calificacion
    
    @staticmethod
    def obtener_rendimiento_estudiante(est_id):
        """Obtiene el rendimiento académico de un estudiante"""
        inscripcion = Inscripcion.query.filter_by(est_id=est_id, inscrito=True).first()
        if not inscripcion:
            return None
        
        calificaciones = Calificacion.query.filter_by(ins_id=inscripcion.id).all()
        
        if not calificaciones:
            return None
        
        promedio_general = sum(c.nota_final for c in calificaciones) / len(calificaciones)
        materias_aprobadas = sum(1 for c in calificaciones if c.aprobado)
        materias_reprobadas = len(calificaciones) - materias_aprobadas
        
        return {
            'promedio_general': round(promedio_general, 2),
            'materias_aprobadas': materias_aprobadas,
            'materias_reprobadas': materias_reprobadas,
            'total_materias': len(calificaciones),
            'calificaciones': calificaciones
        }

"""
Servicios de lógica de negocio para el módulo financiero
"""
from app import db
from app.models import Pago, Inscripcion, Costo
from decimal import Decimal

class FinancieroService:
    
    @staticmethod
    def generar_plan_pagos(ins_id):
        """Genera el plan de pagos para una inscripción"""
        inscripcion = Inscripcion.query.get_or_404(ins_id)
        costo = Costo.query.filter_by(cur_id=inscripcion.cur_id).first()
        
        if not costo:
            raise ValueError("No se encontró configuración de costos para este curso")
        
        # Eliminar pagos previos si existen
        Pago.query.filter_by(ins_id=ins_id).delete()
        
        # Calcular cuota con descuento
        descuento_decimal = Decimal(inscripcion.descu) / Decimal(100)
        cuota_final = costo.cuota * (Decimal(1) - descuento_decimal)
        
        # Generar cuotas
        pagos_creados = []
        for i in range(1, costo.nro_cuota + 1):
            pago = Pago(
                ins_id=ins_id,
                nro_cuota=i,
                cuota=float(cuota_final),
                pagado=False
            )
            db.session.add(pago)
            pagos_creados.append(pago)
        
        db.session.commit()
        return pagos_creados
    
    @staticmethod
    def registrar_pago(pago_id, metodo_pago, nro_comprobante=None):
        """Registra un pago realizado"""
        pago = Pago.query.get_or_404(pago_id)
        
        if pago.pagado:
            raise ValueError("Esta cuota ya ha sido pagada")
        
        pago.pagado = True
        pago.metodo_pago = metodo_pago
        pago.nro_comprobante = nro_comprobante
        
        db.session.commit()
        return pago
    
    @staticmethod
    def obtener_estado_cuenta(ins_id):
        """Obtiene el estado de cuenta de una inscripción"""
        pagos = Pago.query.filter_by(ins_id=ins_id).order_by(Pago.nro_cuota).all()
        
        total_cuotas = len(pagos)
        pagadas = sum(1 for p in pagos if p.pagado)
        pendientes = total_cuotas - pagadas
        monto_total = sum(p.cuota for p in pagos)
        monto_pagado = sum(p.cuota for p in pagos if p.pagado)
        monto_pendiente = monto_total - monto_pagado
        
        return {
            'total_cuotas': total_cuotas,
            'cuotas_pagadas': pagadas,
            'cuotas_pendientes': pendientes,
            'monto_total': monto_total,
            'monto_pagado': monto_pagado,
            'monto_pendiente': monto_pendiente,
            'porcentaje_pagado': (monto_pagado / monto_total * 100) if monto_total > 0 else 0,
            'pagos': pagos
        }

"""
Decoradores para control de acceso basado en roles y permisos
"""
from functools import wraps
from flask import abort, flash, redirect, url_for
from flask_login import current_user

def permiso_requerido(*permisos):
    """Decorador que verifica si el usuario tiene alguno de los permisos especificados"""
    def decorator(f):
        @wraps(f)
        def decorated_function(*args, **kwargs):
            if not current_user.is_authenticated:
                flash('Debe iniciar sesión para acceder', 'warning')
                return redirect(url_for('auth.login'))
            
            if not any(current_user.tiene_permiso(p) for p in permisos):
                flash('No tiene permisos para acceder a esta sección', 'danger')
                abort(403)
            
            return f(*args, **kwargs)
        return decorated_function
    return decorator

def rol_requerido(*roles):
    """Decorador que verifica si el usuario tiene alguno de los roles especificados"""
    def decorator(f):
        @wraps(f)
        def decorated_function(*args, **kwargs):
            if not current_user.is_authenticated:
                flash('Debe iniciar sesión para acceder', 'warning')
                return redirect(url_for('auth.login'))
            
            if not any(current_user.tiene_rol(r) for r in roles):
                flash('No tiene el rol necesario para acceder', 'danger')
                abort(403)
            
            return f(*args, **kwargs)
        return decorated_function
    return decorator
