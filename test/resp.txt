Resumen de entidades principales
- users (gestión de acceso, personal, estudiantes)
- roles, user_roles, permissions, role_permissions
- students, teachers, staff (perfiles extendidos)
- academic_years, terms/semesters
- courses, course_sections (oferta por periodo), course_assignments (profesores asignados)
- enrollments (inscripciones de estudiantes a secciones)
- attendance_records
- grades, grade_items (por actividad), grade_components
- payments, payment_methods
- classrooms
- subjects (materias / asignaturas)
- schedules (horarios por sección)
- audit_logs

Script SQL (Postgres). Ejecuta en orden en una DB vacía.

BEGIN SCRIPT
-- 1. Extensiones útiles
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

-- 2. Tipos enumerados
CREATE TYPE user_status AS ENUM ('active','inactive','suspended');
CREATE TYPE gender AS ENUM ('male','female','other','unknown');
CREATE TYPE payment_status AS ENUM ('pending','paid','failed','refunded');
CREATE TYPE attendance_status AS ENUM ('present','absent','late','excused');

-- 3. Tablas de autenticación / usuarios / roles / permisos
CREATE TABLE roles (
  id SERIAL PRIMARY KEY,
  name TEXT UNIQUE NOT NULL,
  description TEXT
);

CREATE TABLE permissions (
  id SERIAL PRIMARY KEY,
  name TEXT UNIQUE NOT NULL,
  description TEXT
);

CREATE TABLE users (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  username TEXT UNIQUE NOT NULL,
  email TEXT UNIQUE,
  password_hash TEXT NOT NULL,
  full_name TEXT,
  status user_status DEFAULT 'active',
  created_at TIMESTAMP WITH TIME ZONE DEFAULT now(),
  last_login TIMESTAMP WITH TIME ZONE
);

CREATE TABLE user_roles (
  user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  role_id INT NOT NULL REFERENCES roles(id) ON DELETE CASCADE,
  assigned_at TIMESTAMP WITH TIME ZONE DEFAULT now(),
  PRIMARY KEY (user_id, role_id)
);

CREATE TABLE role_permissions (
  role_id INT NOT NULL REFERENCES roles(id) ON DELETE CASCADE,
  permission_id INT NOT NULL REFERENCES permissions(id) ON DELETE CASCADE,
  PRIMARY KEY (role_id, permission_id)
);

-- 4. Perfiles académicos
CREATE TABLE students (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  user_id UUID UNIQUE NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  student_number TEXT UNIQUE,
  birth_date DATE,
  gender gender,
  enrollment_date DATE,
  status user_status DEFAULT 'active'
);

CREATE TABLE teachers (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  user_id UUID UNIQUE NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  teacher_number TEXT UNIQUE,
  hire_date DATE,
  specialization TEXT,
  status user_status DEFAULT 'active'
);

CREATE TABLE staff (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  user_id UUID UNIQUE NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  position TEXT,
  hire_date DATE,
  status user_status DEFAULT 'active'
);

-- 5. Académico: años, términos, materias, cursos
CREATE TABLE academic_years (
  id SERIAL PRIMARY KEY,
  name TEXT UNIQUE NOT NULL, -- e.g. '2025-2026'
  start_date DATE,
  end_date DATE,
  is_active BOOLEAN DEFAULT FALSE
);

CREATE TABLE terms (
  id SERIAL PRIMARY KEY,
  academic_year_id INT NOT NULL REFERENCES academic_years(id) ON DELETE CASCADE,
  name TEXT NOT NULL, -- e.g. 'Term 1', 'Semester 2'
  start_date DATE,
  end_date DATE,
  UNIQUE(academic_year_id, name)
);

CREATE TABLE subjects (
  id SERIAL PRIMARY KEY,
  code TEXT UNIQUE NOT NULL,
  name TEXT NOT NULL,
  description TEXT,
  credits INT DEFAULT 0
);

CREATE TABLE courses (
  id SERIAL PRIMARY KEY,
  subject_id INT NOT NULL REFERENCES subjects(id) ON DELETE RESTRICT,
  title TEXT NOT NULL,
  description TEXT,
  level TEXT, -- e.g. 'Primary', 'Secondary', 'Undergrad'
  default_credits INT DEFAULT 0
);

-- 6. Secciones / oferta por término / asignación de profesores
CREATE TABLE course_sections (
  id SERIAL PRIMARY KEY,
  course_id INT NOT NULL REFERENCES courses(id) ON DELETE CASCADE,
  academic_year_id INT NOT NULL REFERENCES academic_years(id) ON DELETE CASCADE,
  term_id INT NOT NULL REFERENCES terms(id) ON DELETE CASCADE,
  section_code TEXT NOT NULL, -- e.g. 'A', '101'
  capacity INT,
  classroom_id INT,
  status TEXT DEFAULT 'open',
  UNIQUE(course_id, term_id, section_code)
);

CREATE TABLE course_assignments (
  id SERIAL PRIMARY KEY,
  section_id INT NOT NULL REFERENCES course_sections(id) ON DELETE CASCADE,
  teacher_id UUID NOT NULL REFERENCES teachers(id) ON DELETE RESTRICT,
  assigned_at TIMESTAMP WITH TIME ZONE DEFAULT now(),
  role_in_section TEXT DEFAULT 'teacher' -- e.g. 'teacher','assistant'
);

-- 7. Horarios y aulas
CREATE TABLE classrooms (
  id SERIAL PRIMARY KEY,
  name TEXT NOT NULL,
  location TEXT,
  capacity INT
);

CREATE TABLE schedules (
  id SERIAL PRIMARY KEY,
  section_id INT NOT NULL REFERENCES course_sections(id) ON DELETE CASCADE,
  weekday SMALLINT NOT NULL, -- 0=Sunday..6=Saturday or 1..7
  start_time TIME NOT NULL,
  end_time TIME NOT NULL,
  classroom_id INT REFERENCES classrooms(id)
);

-- 8. Inscripciones / matrícula
CREATE TABLE enrollments (
  id SERIAL PRIMARY KEY,
  student_id UUID NOT NULL REFERENCES students(id) ON DELETE CASCADE,
  section_id INT NOT NULL REFERENCES course_sections(id) ON DELETE CASCADE,
  enrolled_at TIMESTAMP WITH TIME ZONE DEFAULT now(),
  status TEXT DEFAULT 'enrolled', -- 'enrolled','withdrawn','completed'
  UNIQUE(student_id, section_id)
);

-- 9. Asistencia
CREATE TABLE attendance_records (
  id BIGSERIAL PRIMARY KEY,
  enrollment_id INT NOT NULL REFERENCES enrollments(id) ON DELETE CASCADE,
  date DATE NOT NULL,
  status attendance_status NOT NULL DEFAULT 'present',
  recorded_by UUID REFERENCES users(id),
  recorded_at TIMESTAMP WITH TIME ZONE DEFAULT now(),
  UNIQUE(enrollment_id, date)
);

-- 10. Evaluaciones y calificaciones
CREATE TABLE grade_items (
  id SERIAL PRIMARY KEY,
  section_id INT NOT NULL REFERENCES course_sections(id) ON DELETE CASCADE,
  name TEXT NOT NULL, -- e.g. 'Exam 1', 'Homework set 3'
  max_score NUMERIC(8,2) DEFAULT 100,
  weight NUMERIC(5,4) DEFAULT 0, -- weight in final grade (sum weights per section should be 1.0)
  date DATE
);

CREATE TABLE grades (
  id BIGSERIAL PRIMARY KEY,
  grade_item_id INT NOT NULL REFERENCES grade_items(id) ON DELETE CASCADE,
  enrollment_id INT NOT NULL REFERENCES enrollments(id) ON DELETE CASCADE,
  score NUMERIC(8,2) CHECK (score >= 0),
  recorded_at TIMESTAMP WITH TIME ZONE DEFAULT now(),
  recorded_by UUID REFERENCES users(id),
  UNIQUE(grade_item_id, enrollment_id)
);

-- 11. Pagos / facturación
CREATE TABLE payment_methods (
  id SERIAL PRIMARY KEY,
  name TEXT UNIQUE NOT NULL -- e.g. 'credit_card','bank_transfer','cash'
);

CREATE TABLE payments (
  id BIGSERIAL PRIMARY KEY,
  student_id UUID NOT NULL REFERENCES students(id) ON DELETE CASCADE,
  amount NUMERIC(12,2) NOT NULL CHECK(amount >= 0),
  currency TEXT DEFAULT 'USD',
  method_id INT REFERENCES payment_methods(id),
  status payment_status DEFAULT 'pending',
  reference TEXT,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT now(),
  processed_at TIMESTAMP WITH TIME ZONE
);

-- 12. Auditoría
CREATE TABLE audit_logs (
  id BIGSERIAL PRIMARY KEY,
  user_id UUID REFERENCES users(id),
  action TEXT NOT NULL,
  object_type TEXT,
  object_id TEXT,
  details JSONB,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT now()
);

-- 13. Índices útiles
CREATE INDEX idx_enrollments_student ON enrollments(student_id);
CREATE INDEX idx_enrollments_section ON enrollments(section_id);
CREATE INDEX idx_attendance_enrollment_date ON attendance_records(enrollment_id, date);

-- 14. Procedimientos / triggers de ejemplo
-- a) auto-insert enrollment audit log
CREATE OR REPLACE FUNCTION fn_audit_enrollment() RETURNS TRIGGER AS $$
BEGIN
  INSERT INTO audit_logs(user_id, action, object_type, object_id, details)
  VALUES (current_setting('app.current_user_id', true)::UUID, 'enrollment_' || TG_OP, 'enrollment', NEW.id::TEXT, to_jsonb(NEW));
  RETURN NEW;
EXCEPTION WHEN others THEN
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_audit_enrollment
AFTER INSERT OR UPDATE OR DELETE ON enrollments
FOR EACH ROW EXECUTE FUNCTION fn_audit_enrollment();

-- b) Prevent enrolling if capacity exceeded
CREATE OR REPLACE FUNCTION fn_check_capacity() RETURNS TRIGGER AS $$
DECLARE
  cap INT;
  enrolled_count INT;
BEGIN
  SELECT capacity INTO cap FROM course_sections WHERE id = NEW.section_id;
  IF cap IS NULL THEN
    RETURN NEW;
  END IF;
  SELECT COUNT(*) INTO enrolled_count FROM enrollments WHERE section_id = NEW.section_id AND status = 'enrolled';
  IF TG_OP = 'INSERT' THEN
    enrolled_count := enrolled_count; -- current count without NEW
  ELSIF TG_OP = 'UPDATE' AND OLD.section_id <> NEW.section_id THEN
    -- if changing section, check new section capacity
    SELECT COUNT(*) INTO enrolled_count FROM enrollments WHERE section_id = NEW.section_id AND status = 'enrolled';
  END IF;
  IF enrolled_count >= cap THEN
    RAISE EXCEPTION 'Capacity exceeded for section % (capacity %)', NEW.section_id, cap;
  END IF;
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_check_capacity
BEFORE INSERT OR UPDATE ON enrollments
FOR EACH ROW EXECUTE FUNCTION fn_check_capacity();

-- c) Simple function to compute final grade for an enrollment in a section
CREATE OR REPLACE FUNCTION fn_final_grade_for_enrollment(enrollment_id INT) RETURNS NUMERIC(8,2) AS $$
DECLARE
  s NUMERIC := 0;
  total_weight NUMERIC := 0;
BEGIN
  SELECT SUM( (g.score / gi.max_score) * gi.weight ) INTO s
  FROM grades g
  JOIN grade_items gi ON gi.id = g.grade_item_id
  WHERE g.enrollment_id = enrollment_id;

  SELECT SUM(weight) INTO total_weight FROM grade_items gi JOIN enrollments e ON gi.section_id = e.section_id WHERE e.id = enrollment_id;

  IF total_weight IS NULL OR total_weight = 0 THEN
    RETURN NULL;
  END IF;

  RETURN ROUND((s / total_weight) * 100, 2); -- return percentage 0..100
END;
$$ LANGUAGE plpgsql;

END SCRIPT

Gestión de usuarios, roles y permisos (ejemplos de datos y grants)
-- crear roles base
INSERT INTO roles(name, description) VALUES
('admin','Administrador del sistema'),
('teacher','Profesor'),
('student','Estudiante'),
('finance','Finanzas'),
('staff','Personal');

-- permisos (ejemplos)
INSERT INTO permissions(name, description) VALUES
('manage_users','Crear/editar/Eliminar usuarios'),
('manage_courses','Crear/editar cursos y secciones'),
('record_attendance','Registrar asistencia'),
('record_grades','Registrar calificaciones'),
('view_reports','Ver reportes financieros y académicos'),
('manage_payments','Registrar y procesar pagos');

-- Asocia permisos a roles (ejemplo)
INSERT INTO role_permissions(role_id, permission_id)
SELECT r.id, p.id FROM roles r, permissions p
WHERE (r.name = 'admin') OR
      (r.name = 'teacher' AND p.name IN ('record_attendance','record_grades','view_reports')) OR
      (r.name = 'finance' AND p.name IN ('manage_payments','view_reports'));

Nota: la tabla role_permissions se puede poblar selectivamente. La lógica de autorización a nivel de aplicación debe consultar user_roles + role_permissions.

Sugerencias operativas y de implementación
- Implementa control de acceso en la capa aplicación consultando users->user_roles->role_permissions.
- Usa row-level security (RLS) si quieres controlar acceso directamente en DB por roles de Postgres.
- Agrega índices adicionales en columnas usadas en filtros/reportes (fechas, estados).
- Considera normalizar más o añadir vistas materializadas para reportes de notas finales, carga de estudiantes por sección, y balances de pagos.
- Logs/auditoría para acciones críticas (cambios de notas, pagos) con triggers o la capa aplicación.
